activate VIDREPO #lightblue

deactivateafter VIDREPO

activate AUTHREPO #lightblue

deactivateafter AUTHREPO

deactivateafter ELIG
alt eligible
  CTRL -> GRANT : issue grant
  activate GRANT #lightblue
  GRANT -> AUTHREPO : persist grant
  activate AUTHREPO #lightblue
  AUTHREPO --> GRANT : stored
  deactivateafter AUTHREPO
  GRANT -> AUDIT : record event
  activate AUDIT #lightblue
  AUDIT --> GRANT : event recorded
  deactivateafter AUDIT
  GRANT --> CTRL : grant
  deactivateafter GRANT
  [<--CTRL : return VID and grant
else not eligible
  [<--CTRL : reject request
end

activate CTRL #lightblue

activate VID #lightblue

activate GRANT #lightblue
alt grant already consumed with matching nonce (idempotent replay)
  GRANT --> CTRL : return prior consume success
else grant not yet consumed
  alt grant expired, election/VID mismatch, or already voted
    GRANT --> CTRL : reject (invalid grant)
  else grant valid and unconsumed
    GRANT -> AUTHREPO : atomically update only if unconsumed and unexpired
    activate AUTHREPO #lightblue
    AUTHREPO --> GRANT : consumed or rejected
    deactivateafter AUTHREPO
    GRANT -> AUDIT : record consume outcome
    activate AUDIT #lightblue
    AUDIT --> GRANT : event recorded
    deactivateafter AUDIT
    GRANT --> CTRL : consume response
  end
end
