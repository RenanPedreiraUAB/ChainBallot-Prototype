title Internal Sequence: Voter Component

participantgroup Voter Component (Internal)
participant "Voter Interaction Interface" as UI #lightblue
participant "Ballot Preparation Service" as PREP #lightblue
participant "Encryption Service" as CRYPTO #lightblue
participant "Submission Outbox" as OUTBOX #lightblue
end

participantgroup External Participants
participant "Auth Provider Client" as AUTH #lightblue
participant "Election Authorization Client" as AUTHZ #lightblue
participant "Metadata Client" as META #lightblue
participant "Queue Publisher" as QUEUE #lightblue
end

[->UI : authenticate and open ballot page
activate UI #lightblue
UI -> AUTH : start auth flow
activate AUTH #lightblue
AUTH --> UI : signed assertion
deactivateafter AUTH
UI -> AUTHZ : load VID profile
activate AUTHZ #lightblue
AUTHZ --> UI : VID profile
deactivateafter AUTHZ
UI -> AUTHZ : load election authorization state
activate AUTHZ #lightblue
AUTHZ --> UI : eligibility state
deactivateafter AUTHZ
UI -> META : fetch election metadata
activate META #lightblue
META --> UI : election context
deactivateafter META
UI -> AUTHZ : request one-time grant
activate AUTHZ #lightblue
AUTHZ --> UI : authorization grant
deactivateafter AUTHZ
UI -> PREP : prepare ballot
activate PREP #lightblue
PREP -> CRYPTO : encrypt vote payload
activate CRYPTO #lightblue
CRYPTO --> PREP : encrypted payload
deactivateafter CRYPTO
PREP -> PREP : compute ballot commitment
activate PREP #lightblue
deactivateafter PREP
PREP --> UI : submission package
deactivateafter PREP
UI -> OUTBOX : durably stage package (before authorization is consumed)
activate OUTBOX #lightblue
OUTBOX --> UI : staged
deactivateafter OUTBOX
alt durable staging fails
  [<--UI : submission rejected (not authorized; retry staging)
else durable staging succeeds
  UI -> AUTHZ : consume grant (grantId + submissionNonce)
  activate AUTHZ #lightblue
  AUTHZ --> UI : consume outcome
  deactivateafter AUTHZ
  alt consume succeeds
    UI -> OUTBOX : mark staged entry authorized (pending publish)
    activate OUTBOX #lightblue
    OUTBOX --> UI : authorized
    deactivateafter OUTBOX
    UI -> QUEUE : publish ballot package
    activate QUEUE #lightblue
    QUEUE --> UI : publish outcome
    deactivateafter QUEUE
    alt publish succeeds
      UI -> OUTBOX : remove confirmed entry
      activate OUTBOX #lightblue
      OUTBOX --> UI : removed
      deactivateafter OUTBOX
      UI -> UI : display submission receipt (not persisted)
      activate UI #lightblue
      deactivateafter UI
      [<--UI : ballot submitted
    else publish fails
      [<--UI : submission queued for retry (already staged and authorized)
    end
  else consume fails
    UI -> OUTBOX : remove staged entry (never authorized; must never be published)
    activate OUTBOX #lightblue
    OUTBOX --> UI : removed
    deactivateafter OUTBOX
    [<--UI : submission rejected
  end
end
deactivateafter UI

[->UI : load dashboard (auto-retries all pending outbox entries)
activate UI #lightblue
loop until outbox empty or a retry failure occurs
  UI -> OUTBOX : take next pending entry
  activate OUTBOX #lightblue
  OUTBOX --> UI : pending entry (Staged or AuthorizedPendingPublish) or none
  deactivateafter OUTBOX
  alt entry found and status is Staged
    UI -> AUTHZ : consume grant using entry's stored grantId
    activate AUTHZ #lightblue
    AUTHZ --> UI : consume outcome
    deactivateafter AUTHZ
    alt consume succeeds
      UI -> UI : treat entry as authorized, continue to publish below
    else consume fails
      UI -> UI : drop entry (never authorized; never published)
    end
  end
  alt entry authorized (already was, or just became so)
    UI -> QUEUE : publish pending package
    activate QUEUE #lightblue
    QUEUE --> UI : retry publish outcome
    deactivateafter QUEUE
    alt retry publish succeeds
      UI -> OUTBOX : remove confirmed entry
      activate OUTBOX #lightblue
      OUTBOX --> UI : removed
      deactivateafter OUTBOX
      UI -> UI : display submission receipt (not persisted)
      activate UI #lightblue
      deactivateafter UI
    else retry publish fails
      UI -> OUTBOX : requeue pending package
      activate OUTBOX #lightblue
      OUTBOX --> UI : retry queued
      deactivateafter OUTBOX
    end
  else no pending entry
    [<--UI : no further pending submissions
  end
end
[<--UI : dashboard ready
deactivateafter UI

[->UI : inspect outbox state
activate UI #lightblue
UI -> OUTBOX : read pending state
activate OUTBOX #lightblue
OUTBOX --> UI : pending entries and stats
deactivateafter OUTBOX
[<--UI : outbox state shown
deactivateafter UI

[->UI : retry next failed submission
activate UI #lightblue
UI -> OUTBOX : take next pending
activate OUTBOX #lightblue
OUTBOX --> UI : pending package or none
deactivateafter OUTBOX
alt pending package found
  UI -> QUEUE : publish queued entry
  activate QUEUE #lightblue
  QUEUE --> UI : manual retry publish outcome
  deactivateafter QUEUE
  alt manual retry succeeds
    UI -> OUTBOX : remove confirmed entry
    activate OUTBOX #lightblue
    OUTBOX --> UI : removed
    deactivateafter OUTBOX
    UI -> UI : display submission receipt (not persisted)
    activate UI #lightblue
    deactivateafter UI
  else manual retry fails
    UI -> OUTBOX : requeue pending package
    activate OUTBOX #lightblue
    OUTBOX --> UI : retry queued
    deactivateafter OUTBOX
  end
else no pending package
  [<--UI : no pending submissions
end
deactivateafter UI

[->UI : compact stale/excess entries
activate UI #lightblue
UI -> OUTBOX : compact outbox
activate OUTBOX #lightblue
OUTBOX --> UI : compact summary
deactivateafter OUTBOX
[<--UI : compaction result shown
deactivateafter UI
