title UIS-02 (1/2) — Voter Casting, Authorization &amp; Queuing


actor "Voter" as VOTER #lightblue

participant "Voter Component" as VC #lightblue
participant "Election Authorization Component" as AUTH #lightblue
database "Voting Authorization Repository" as AUTHDB #lightblue
participant "Message Broker\n(Durable Queue Infrastructure)" as QUEUE #lightblue

activate VOTER #lightblue
activate VC #lightblue

VOTER -> VC : Select candidate to vote
VC --> VOTER: Request candidate confirmation
VOTER -> VC : Submit ballot

VC -> AUTH : Request one-time voting grant

activate AUTH #lightblue

AUTH -> AUTHDB : Validate voting authorization state

activate AUTHDB #lightblue
AUTHDB --> AUTH : Return voting authorization status
deactivateafter AUTHDB

alt Invalid or consumed authorization
AUTH --> VC : Returns error: no authorization to vote
VC --> VOTER : Display submission failure

else Valid authorization

AUTH --> VC : Return authorization grant
deactivateafter AUTH

VC -> VC : Encrypt canonical ballot representation
activate VC #lightblue
deactivateafter VC

VC -> VC : Generate ballot submission package\n(encrypted vote, VID, election identifier,\nballot commitment, submission nonce, ballot nullifier)
activate VC #lightblue
deactivateafter VC

VC -> VC : Durably persist package in submission outbox\n(normal path, BEFORE authorization is consumed)
activate VC #lightblue
deactivateafter VC

alt Durable staging failed
VC --> VOTER : Display submission failure (authorization not consumed)
deactivateafter VC
deactivateafter VOTER

else Durable staging succeeded

VC -> AUTH : Consume authorization grant\n(grant identifier + submission nonce)
activate AUTH #lightblue
AUTH -> AUTHDB : Atomically consume vote authorization
activate AUTHDB #lightblue
AUTHDB --> AUTH : Confirm update operation
deactivateafter AUTHDB
AUTH --> VC : Confirm authorization consumption
deactivateafter AUTH

VC -> QUEUE : Publish encrypted ballot package
activate QUEUE #lightblue
QUEUE --> VC : Confirm broker acknowledgement
deactivateafter QUEUE

VC -> VC : Confirm publication and remove outbox entry\n(reconciles staged package with confirmed publish)
activate VC #lightblue
deactivateafter VC

VC --> VOTER : Confirm ballot submission

deactivateafter VC
deactivateafter VOTER

end

end
